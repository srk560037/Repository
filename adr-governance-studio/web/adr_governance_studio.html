// Phase 1 Core Engine: Go CLI Binary for Parsing & Linting ADR Markdown Files
package main

import (
	"fmt"
	"io/ioutil"
	"os"
	"path/filepath"
	"strings"
	"gopkg.in/yaml.v3"
)

type ADRFrontmatter struct {
	ID              string   `yaml:"id"`
	Title           string   `yaml:"title"`
	Status          string   `yaml:"status"`
	Date            string   `yaml:"date"`
	Authors         []string `yaml:"authors"`
	Tags            []string `yaml:"tags"`
	ComplianceRules []struct {
		RuleID      string `yaml:"rule_id"`
		Description string `yaml:"description"`
	} `yaml:"compliance_rules"`
}

func main() {
	if len(os.Args) < 2 {
		fmt.Println("Usage: adr  [args]")
		os.Exit(1)
	}

	cmd := os.Args[1]
	switch cmd {
	case "init":
		os.MkdirAll("docs/architecture/decisions", 0755)
		fmt.Println("✔ Initialized ADR repository structure at ./docs/architecture/decisions")
	case "lint":
		lintADRFiles()
	default:
		fmt.Printf("Unknown command: %s\n", cmd)
	}
}

func lintADRFiles() {
	files, _ := filepath.Glob("docs/architecture/decisions/*.md")
	fmt.Printf("Linting %d ADR Markdown documents...\n", len(files))
	for _, file := range files {
		content, _ := ioutil.ReadFile(file)
		parts := strings.Split(string(content), "---")
		if len(parts) < 3 {
			fmt.Printf("❌ %s: Missing YAML frontmatter bounds\n", file)
			continue
		}
		var meta ADRFrontmatter
		err := yaml.Unmarshal([]byte(parts[1]), &meta)
		if err != nil || meta.ID == "" {
			fmt.Printf("❌ %s: Invalid YAML Schema parsing\n", file)
		} else {
			fmt.Printf("✔ %s [%s]: Schema Valid\n", file, meta.ID)
		}
	}
}

# Phase 4: Python FastAPI RAG Vector Search Engine & AI Auto-Drafter
import os
from fastapi import FastAPI, HTTPException
from pydantic import BaseModel
from langchain_community.vectorstores import Qdrant
from langchain_openai import OpenAIEmbeddings, ChatOpenAI
from langchain.chains import RetrievalQA

app = FastAPI(title="ADR Architecture AI RAG Service")

class QueryRequest(BaseModel):
    prompt: str

class DraftRequest(BaseModel):
    raw_notes: str

@app.post("/api/v1/query-adr")
async def query_architecture_kb(req: QueryRequest):
    embeddings = OpenAIEmbeddings(model="text-embedding-3-small")
    vector_store = Qdrant.from_existing_collection(
        embedding=embeddings,
        collection_name="enterprise_adrs"
    )
    
    llm = ChatOpenAI(model="gpt-4o", temperature=0.1)
    qa_chain = RetrievalQA.from_chain_type(llm=llm, retriever=vector_store.as_retriever())
    
    response = qa_chain.run(req.prompt)
    return {"query": req.prompt, "answer": response}

@app.post("/api/v1/auto-draft-adr")
async def auto_draft_adr(req: DraftRequest):
    llm = ChatOpenAI(model="gpt-4o", temperature=0.2)
    system_prompt = (
        "You are an Enterprise Principal Architect. Convert the following meeting notes "
        "into a valid YAML Frontmatter + Markdown Architecture Decision Record (ADR)."
    )
    result = llm.invoke(f"{system_prompt}\n\nNotes:\n{req.raw_notes}")
    return {"markdown_adr": result.content}
